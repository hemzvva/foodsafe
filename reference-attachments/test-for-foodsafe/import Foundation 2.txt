import Foundation

enum IngredientRole: String, Sendable {
    /// The dish doesn't work without it.
    case core
    /// Finishing touches. Missing ones make a recipe "Almost Ready".
    case garnish
    /// Salt, pepper and spices, used to taste.
    case seasoning

    var isOptional: Bool { self != .core }
}

struct RecipeItem: Hashable, Sendable {
    let ingredientID: String
    /// Amount for the recipe's base servings, in the ingredient's base unit.
    let amount: Double
    let role: IngredientRole
}

struct RecipeStep: Hashable, Sendable {
    let title: String
    /// Text with `{ingredient_id}` tokens, filled with scaled amounts at runtime.
    let text: String
    let timerMinutes: Int?
}

struct Recipe: Identifiable, Hashable, Sendable {
    let id: String
    let name: String
    let summary: String
    let minutes: Int
    let servings: Int
    let ingredients: [RecipeItem]
    let steps: [RecipeStep]
    let tip: String

    var art: String { "dish_\(id)" }
    var coreItems: [RecipeItem] { ingredients.filter { $0.role == .core } }

    func uses(_ ingredientID: String) -> Bool { ingredients.contains { $0.ingredientID == ingredientID } }
}

private func c(_ id: String, _ amount: Double) -> RecipeItem { RecipeItem(ingredientID: id, amount: amount, role: .core) }
private func g(_ id: String, _ amount: Double) -> RecipeItem { RecipeItem(ingredientID: id, amount: amount, role: .garnish) }
private func s(_ id: String, _ amount: Double) -> RecipeItem { RecipeItem(ingredientID: id, amount: amount, role: .seasoning) }
private func step(_ title: String, _ text: String, timer: Int? = nil) -> RecipeStep {
    RecipeStep(title: title, text: text, timerMinutes: timer)
}
