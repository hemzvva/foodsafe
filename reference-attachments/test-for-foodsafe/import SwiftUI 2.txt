import SwiftUI

/// Pen-on-paper path builder working on a 24 × 24 grid. Straight lines bow
/// very slightly, the way a real pen stroke does.
struct GlyphPen {
    private(set) var path = Path()
    private var strokeIndex = 0

    mutating func m(_ x: CGFloat, _ y: CGFloat) {
        path.move(to: CGPoint(x: x, y: y))
    }

    mutating func l(_ x: CGFloat, _ y: CGFloat) {
        let start = path.currentPoint ?? .zero
        let end = CGPoint(x: x, y: y)
        let dx = end.x - start.x, dy = end.y - start.y
        let length = max((dx * dx + dy * dy).squareRoot(), 0.001)
        strokeIndex += 1
        let bow = (strokeIndex.isMultiple(of: 2) ? 1 : -1) * min(length * 0.025, 0.35)
        let control = CGPoint(x: (start.x + end.x) / 2 - dy / length * bow,
                              y: (start.y + end.y) / 2 + dx / length * bow)
        path.addQuadCurve(to: end, control: control)
    }

    mutating func q(_ cx: CGFloat, _ cy: CGFloat, _ x: CGFloat, _ y: CGFloat) {
        path.addQuadCurve(to: CGPoint(x: x, y: y), control: CGPoint(x: cx, y: cy))
    }

    mutating func c(_ c1x: CGFloat, _ c1y: CGFloat, _ c2x: CGFloat, _ c2y: CGFloat, _ x: CGFloat, _ y: CGFloat) {
        path.addCurve(to: CGPoint(x: x, y: y), control1: CGPoint(x: c1x, y: c1y), control2: CGPoint(x: c2x, y: c2y))
    }

    mutating func close() { path.closeSubpath() }

    mutating func circle(_ cx: CGFloat, _ cy: CGFloat, _ r: CGFloat) {
        path.addPath(SketchCircle(seed: UInt64(cx * 10 + cy)).path(in: CGRect(x: cx - r, y: cy - r, width: r * 2, height: r * 2)))
    }

    mutating func dot(_ cx: CGFloat, _ cy: CGFloat) {
        path.addEllipse(in: CGRect(x: cx - 0.55, y: cy - 0.55, width: 1.1, height: 1.1))
    }

    mutating func rect(_ x: CGFloat, _ y: CGFloat, _ w: CGFloat, _ h: CGFloat, _ r: CGFloat) {
        path.addPath(SketchRect(radius: r, seed: UInt64(x * 7 + y * 3 + w), wobble: 0.14)
            .path(in: CGRect(x: x, y: y, width: w, height: h)))
    }
}

struct GlyphShape: Shape {
    let glyph: Glyph

    func path(in rect: CGRect) -> Path {
        var pen = GlyphPen()
        glyph.draw(&pen)
        let s = min(rect.width, rect.height) / 24
        let transform = CGAffineTransform(scaleX: s, y: s)
            .concatenating(CGAffineTransform(translationX: rect.midX - 12 * s, y: rect.midY - 12 * s))
        return pen.path.applying(transform)
    }
}

/// An icon drawn in ink, with a faint second pass for a pencil-on-paper feel.
struct GlyphView: View {
    let glyph: Glyph
    var size: CGFloat = 22
    var weight: CGFloat = 1.8
    @ScaledMetric(relativeTo: .body) private var scale = 1.0

    var body: some View {
        let side = size * min(scale, 1.6)
        let width = weight * side / 22
        ZStack {
            if glyph.isFilled {
                GlyphShape(glyph: glyph).fill()
            } else {
                GlyphShape(glyph: glyph)
                    .stroke(style: StrokeStyle(lineWidth: width, lineCap: .round, lineJoin: .round))
                GlyphShape(glyph: glyph)
                    .stroke(style: StrokeStyle(lineWidth: width * 0.45, lineCap: .round, lineJoin: .round))
                    .offset(x: side * 0.025, y: side * 0.02)
                    .opacity(0.35)
            }
        }
        .frame(width: side, height: side)
        .accessibilityHidden(true)
    }
}

/// A glyph inside a tinted, hand-drawn circle, used for overview rows and settings.
struct GlyphBadge: View {
    let glyph: Glyph
    var tint: Color = Palette.green
    var wash: Color = Palette.greenWash
    var size: CGFloat = 44

    var body: some View {
        GlyphView(glyph: glyph, size: size * 0.5)
            .foregroundStyle(tint)
            .frame(width: size, height: size)
            .background(SketchCircle(seed: stableSeed(glyph.rawValue)).fill(wash))
            .overlay(SketchCircle(seed: stableSeed(glyph.rawValue) &+ 3).stroke(tint.opacity(0.35), lineWidth: 1))
    }
}
