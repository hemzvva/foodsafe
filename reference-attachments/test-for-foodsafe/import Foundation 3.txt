import Foundation

enum Zone: String, CaseIterable, Codable, Identifiable, Sendable {
    case fridge, pantry, freezer

    var id: String { rawValue }

    var title: String {
        switch self {
        case .fridge: "Fridge"
        case .pantry: "Pantry"
        case .freezer: "Freezer"
        }
    }

    var glyph: Glyph {
        switch self {
        case .fridge: .fridge
        case .pantry: .pantry
        case .freezer: .freezer
        }
    }
}

enum FoodCategory: String, CaseIterable, Codable, Identifiable, Sendable {
    case vegetables, fruits, dairy, grains, proteins, others

    var id: String { rawValue }

    var title: String {
        switch self {
        case .vegetables: "Vegetables"
        case .fruits: "Fruits"
        case .dairy: "Dairy"
        case .grains: "Grains"
        case .proteins: "Proteins"
        case .others: "Others"
        }
    }

    var art: String { "cat_\(rawValue)" }
}

/// What an ingredient is measured in. Amounts are always stored in the base unit:
/// grams, millilitres or pieces.
enum Dimension: String, Codable, Sendable {
    case mass, volume, count
}

enum EntryUnit: String, CaseIterable, Identifiable, Sendable {
    case g, kg, ml, l, pcs

    var id: String { rawValue }

    var dimension: Dimension {
        switch self {
        case .g, .kg: .mass
        case .ml, .l: .volume
        case .pcs: .count
        }
    }

    /// Multiply an entered value by this to get base units.
    var factor: Double {
        switch self {
        case .kg, .l: 1000
        default: 1
        }
    }

    var label: String {
        switch self {
        case .g: "g"
        case .kg: "kg"
        case .ml: "ml"
        case .l: "L"
        case .pcs: "pcs"
        }
    }

    static func units(for dimension: Dimension) -> [EntryUnit] {
        allCases.filter { $0.dimension == dimension }
    }
}

struct IngredientInfo: Identifiable, Hashable, Sendable {
    let id: String
    /// Display name, plural for counted things ("Eggs", "Onions").
    let name: String
    /// Singular for counted things, used in sentences ("1 onion").
    let one: String?
    let category: FoodCategory
    let zone: Zone
    let dimension: Dimension
    /// Typical shelf life in days, per storage zone.
    let fridgeDays: Int
    let pantryDays: Int
    let freezerDays: Int
    /// Noun for counted amounts ("clove", "stalk"); nil prints "pcs".
    let countNoun: String?

    var art: String { id }

    func shelfLife(in zone: Zone) -> Int {
        switch zone {
        case .fridge: fridgeDays
        case .pantry: pantryDays
        case .freezer: freezerDays
        }
    }

    var defaultUnit: EntryUnit {
        switch dimension {
        case .mass: .g
        case .volume: .ml
        case .count: .pcs
        }
    }

    /// The word to use in step text: "basil", "onion", "eggs".
    func word(for amount: Double) -> String {
        if dimension == .count, let one, amount <= 1.0001 { return one }
        return name.lowercased()
    }
}

    static func info(_ id: String) -> IngredientInfo? { byID[id] }

    static func name(_ id: String) -> String { byID[id]?.name ?? id }

    /// Best catalog match for free text, used when someone types a grocery name.
    static func search(_ text: String) -> [IngredientInfo] {
        let query = text.trimmingCharacters(in: .whitespaces).lowercased()
        guard !query.isEmpty else { return all }
        return all
            .filter { $0.name.lowercased().contains(query) || ($0.one?.contains(query) ?? false) }
            .sorted { lhs, rhs in
                let l = lhs.name.lowercased().hasPrefix(query), r = rhs.name.lowercased().hasPrefix(query)
                return l != r ? l : lhs.name < rhs.name
            }
    }
