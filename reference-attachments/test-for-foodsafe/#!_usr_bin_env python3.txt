def call(prompt, aspect):
    body = json.dumps({"model": MODEL, "prompt": prompt, "aspect_ratio": aspect}).encode()
    req = urllib.request.Request(URL, data=body, headers={"Authorization": f"Bearer {KEY}", "Content-Type": "application/json", "User-Agent": "foodsafe-artgen/1.0"})
    with urllib.request.urlopen(req, timeout=240) as r:
        d = json.load(r)
    return Image.open(io.BytesIO(base64.b64decode(d["data"][0]["b64_json"]))).convert("RGB")


def color_to_alpha(img):
    """Key the paper colour out so ink and watercolour washes keep a soft edge."""
    w, h = img.size
    strips = [img.crop((0, 0, w, 6)), img.crop((0, h - 6, w, h)), img.crop((0, 0, 6, h)), img.crop((w - 6, 0, w, h))]
    px = [c for st in strips for c in st.getdata()]
    bg = tuple(sorted(c[i] for c in px)[len(px) // 2] for i in range(3))
    diff = ImageChops.difference(img, Image.new("RGB", img.size, bg))
    r, g, b = diff.split()
    m = ImageChops.lighter(ImageChops.lighter(r, g), b)
    # distance from paper -> alpha: ignore grain under 14, full opacity by ~70
    alpha = m.point(lambda v: 0 if v < 14 else min(255, int((v - 14) * 255 / 56)))
    alpha = alpha.filter(ImageFilter.GaussianBlur(0.6))
    # un-premultiply against the paper so light washes don't look milky on dark cards
    def unmix(ch, bgv, a):
        return Image.merge("L", [ch]) if False else ch
    im = img.convert("RGBA")
    im.putalpha(alpha)
    box = alpha.point(lambda v: 255 if v > 40 else 0).getbbox()
    if box:
        x0, y0, x1, y1 = box
        side = max(x1 - x0, y1 - y0)
        half = side // 2 + int(side * 0.06)
        cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
        canvas = Image.new("RGBA", (half * 2, half * 2), (0, 0, 0, 0))
        canvas.paste(im.crop((cx - half, cy - half, cx + half, cy + half)), (0, 0))
        im = canvas
    return im


def write_set(name, img, ext):
    d = os.path.join(ROOT, f"{name}.imageset")
    os.makedirs(d, exist_ok=True)
    fn = f"{name}.{ext}"
    if ext == "png":
        img.save(os.path.join(d, fn), optimize=True)
    else:
        img.save(os.path.join(d, fn), quality=82, optimize=True, progressive=True)
    json.dump({"images": [{"filename": fn, "idiom": "universal"}], "info": {"author": "xcode", "version": 1}},
              open(os.path.join(d, "Contents.json"), "w"))


def job(name, prompt, kind):
    for attempt in range(3):
        try:
            if kind == "cut":
                im = color_to_alpha(call(CUT.format(s=prompt), "1:1"))
                im.thumbnail((360, 360), Image.LANCZOS)
                write_set(name, im, "png")
            elif kind == "spot":
                im = color_to_alpha(call(CUT.format(s=prompt), "1:1"))
                im.thumbnail((720, 720), Image.LANCZOS)
                write_set(name, im, "png")
            elif kind == "dish":
                im = call(DISH.format(s=prompt), "4:3")
                im.thumbnail((960, 720), Image.LANCZOS)
                write_set(name, im, "jpg")
            else:
                im = call(prompt, "1:1")
                im.thumbnail((768, 768), Image.LANCZOS)
                write_set(name, im, "jpg")
            return f"ok {name}"
        except Exception as e:  # retry transient failures
            err = e
    return f"FAIL {name}: {err}"


def main():
    os.makedirs(ROOT, exist_ok=True)
    json.dump({"info": {"author": "xcode", "version": 1}, "properties": {"provides-namespace": False}},
              open(os.path.join(ROOT, "Contents.json"), "w"))
    jobs = [(k, v, "cut") for k, v in INGREDIENTS.items()] + [(k, v, "cut") for k, v in CATEGORIES.items()] \
         + [(k, v, "spot") for k, v in SPOTS.items()] + [(k, v, "dish") for k, v in DISHES.items()] \
         + [(k, v, "tex") for k, v in TEXTURE.items()]
    only = set(sys.argv[1:])
    if only:
        jobs = [j for j in jobs if j[0] in only]
    else:
        jobs = [j for j in jobs if not os.path.isdir(os.path.join(ROOT, f"{j[0]}.imageset"))]
    with cf.ThreadPoolExecutor(8) as ex:
        for r in ex.map(lambda j: job(*j), jobs):
            print(r, flush=True)


if __name__ == "__main__":
    main()
