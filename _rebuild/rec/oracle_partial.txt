49	# ----------------------------------------------------------------------------- units (B-06)
50	KIND = {"g": "mass", "ml": "volume", "L": "volume", "pcs": "piece", "cloves": "clove"}
51	TO_BASE = {"g": F(1), "ml": F(1), "L": F(1000), "pcs": F(1), "cloves": F(1)}
52	
53	
54	def comparable(a: str, b: str) -> bool:
55	    return KIND[a] == KIND[b]
56	
57	
58	def convert(q: F, u_from: str, u_to: str) -> F:
59	    assert comparable(u_from, u_to), (u_from, u_to)
60	    return q * TO_BASE[u_from] / TO_BASE[u_to]
61	
62	
63	def rhu(x: F) -> int:
64	    """Round half up (ties go up). Only non-negative values are used."""
65	    return math.floor(x + F(1, 2))
66	
67	
68	def q2(x: F) -> F:
69	    """Keep at most two decimals (round half up)."""
70	    return F(rhu(x * 100), 100)
71	
72	
73	def dec(v: F) -> str:
74	    """Exact decimal string of a fraction whose denominator divides a power of ten."""
75	    n, d = v.numerator, v.denominator
76	    for k in range(0, 7):
77	        if (n * 10 ** k) % d == 0:
78	            x = n * 10 ** k // d
79	            s = str(abs(x)).rjust(k + 1, "0")
80	            ip, fp = (s[:-k], s[-k:]) if k else (s, "")
81	            return ("-" if x < 0 else "") + ip + ("." + fp if k else "")
82	    raise ValueError("not a short decimal: %s" % v)
83	
84	
85	# ----------------------------------------------------------------------------- B-05 writing an amount
86	def _trim(n: int, digits: int) -> str:
87	    whole, frac = divmod(n, 10 ** digits)
88	    s = "%d.%0*d" % (whole, digits, frac)
89	    return s.rstrip("0").rstrip(".")
90	
91	
92	def number_text(v: F, unit: str) -> str:
93	    if unit in ("g", "ml"):
94	        if v >= 10:
95	            return str(rhu(v))
96	        return _trim(rhu(v * 10), 1)
97	    if unit == "L":
98	        return _trim(rhu(v * 100), 2)
99	    if unit in ("pcs", "cloves"):
100	        n = rhu(v * 2)
101	        whole, odd = divmod(n, 2)
102	        return "%d.5" % whole if odd else str(whole)
103	    raise ValueError(unit)
104	
105	
106	def amount_text(v: F, unit: str) -> str:
107	    return "%s %s" % (number_text(v, unit), unit)
108	
109	
110	def displayed(v: F, unit: str) -> F:
111	    """The numeric value that B-05 writes (what the person reads)."""
112	    if unit in ("g", "ml"):
113	        return F(rhu(v)) if v >= 10 else F(rhu(v * 10), 10)
114	    if unit == "L":
115	        return F(rhu(v * 100), 100)
116	    return F(rhu(v * 2), 2)

257	# ----------------------------------------------------------------------------- B-10 substitute amount
258	def substitute_amount(content: Content, from_ing: str, from_unit: str, amount: F, sub: str) -> Optional[Tuple[F, str]]:
259	    """B-10. None when the units differ in kind and no factor exists (invariant I-10 makes that unreachable with valid content)."""
260	    if KIND[from_unit] == KIND[content.ings[sub].unit]:
261	        return amount, from_unit
262	    found = content.factors.get((from_ing, from_unit, sub))
263	    if found is None:
264	        return None
265	    per, to_unit = found
266	    return amount * per, to_unit

300	def match_recipe(content: Content, r: Recipe, servings: int, items: List[Item], today: date) -> Match:
301	    status: List[Tuple[str, Optional[str]]] = []
302	    for ln in r.lines:
303	        amt = scaled(ln.amount, servings, r.base)
304	        if available(ln.ing, amt, ln.unit, items, today):
305	            status.append(("have", None))
306	            continue
307	        if ln.optional:
308	            status.append(("missingOptional", None))
309	            continue
310	        found = None
311	        members = content.cluster_of(ln.ing)
312	        if members:
313	            for s in members:
314	                if s == ln.ing:
315	                    continue
316	                sub_amt = substitute_amount(content, ln.ing, ln.unit, amt, s)
317	                if sub_amt is None:
318	                    continue  # no factor: this member is not offered
319	                samt, sunit = sub_amt
320	                if available(s, samt, sunit, items, today):
321	                    found = s
322	                    break
323	        status.append(("swap", found) if found else ("missing", None))
324	    n_missing = sum(1 for s, _ in status if s == "missing")
325	    n_swap = sum(1 for s, _ in status if s == "swap")
326	    # D-X-69: an optional line never changes the tier; only required lines decide it
327	    if n_missing == 0 and n_swap == 0:
328	        tier = "readyNow"
329	    elif n_missing == 0 and n_swap >= 1:
330	        tier = "smartSwap"
331	    elif 1 <= n_missing <= K_MISSING:
332	        tier = "almostReady"
333	    else:
334	        tier = None
335	    uf = use_first_ingredients(items, today)
336	    eff: List[str] = []
337	    for (s, sub), ln in zip(status, r.lines):
338	        if s == "have":
339	            eff.append(ln.ing)
340	        elif s == "swap":
341	            eff.append(sub)
342	    rescue = [e for e in dict.fromkeys(eff) if e in uf]
343	    return Match(r, servings, status, tier, rescue)

540	    """S-16: `You have 450 g`. An amount proposal states its stock in the unit of the line; a presence proposal states what the

547	def propose_deductions(content: Content, r: Recipe, servings: int, swaps: Dict[int, str],
548	                       items: List[Item], today: date) -> List[Proposal]:
549	    out: List[Proposal] = []
550	    for i, ln in enumerate(r.lines):
551	        base_amt = scaled(ln.amount, servings, r.base)
552	        if i in swaps:
553	            e = swaps[i]
554	            sub_amt = substitute_amount(content, ln.ing, ln.unit, base_amt, e)
555	            if sub_amt is None:
556	                continue  # no factor: the line gets no deduction
557	            amt, unit = sub_amt
558	        else:
559	            e, amt, unit = ln.ing, base_amt, ln.unit
560	        line_amt = displayed(amt, unit)
561	        rows = sorted(rows_of(e, items, today), key=lambda x: (x.expiry, x.added, x.id))
562	        if not rows:
563	            continue
564	        if any(not comparable(x.unit, unit) for x in rows):
565	            out.append(Proposal(i, e, "presence", F(0), unit, F(0), rows[0].id))
566	            continue
567	        stock = sum((convert(x.qty, x.unit, unit) for x in rows), F(0))
568	        out.append(Proposal(i, e, "amount", min(line_amt, stock), unit, stock))
569	    return out

572	def apply_deductions(items: List[Item], proposals: List[Proposal], presence_on: List[str], today: date) -> Tuple[List[Item], List[Event]]:
573	    rows = list(items)
574	    events: List[Event] = []
575	
576	    def remove(item: Item):
577	        events.append(Event("used", "cook", item.ingredient, item.name, item.qty, item.unit, item.zone, days_left(item.expiry, today)))
578	        rows.remove(item)
579	
580	    for p in proposals:
581	        if p.kind == "presence":
582	            if p.row in presence_on:
583	                gone = next((x for x in rows if x.id == p.row), None)  # an earlier proposal may have removed it
584	                if gone is not None:
585	                    remove(gone)
586	            continue
587	        remaining = p.amount
588	        order = sorted(rows_of(p.ingredient, rows, today), key=lambda x: (x.expiry, x.added, x.id))
589	        for row in order:
590	            if remaining <= 0:
591	                break
592	            remaining_in_row_unit = convert(remaining, p.unit, row.unit)
593	            take = min(remaining_in_row_unit, row.qty)
594	            new_qty = q2(row.qty - take)
595	            remaining -= convert(take, row.unit, p.unit)
596	            idx = rows.index(row)
597	            if new_qty == 0:
598	                remove(row)
599	            else:
600	                rows[idx] = replace(row, qty=new_qty)
601	    return rows, events

605	def digest_schedule(items: List[Item], today: date, now_minute: int, reminder_minute: int = 1080) -> List[Tuple[str, str, str]]:
606	    out = []
607	    for k in range(14):
608	        d = today + timedelta(days=k)

2849	def update_docs() -> int:
2850	    """Replace every generated block of the documents with the text this file produces now. Returns the number of blocks changed."""
2851	    eng = os.path.join(KIT, "03-ENGINEERING")
2852	    jobs = [("TESTING.md", BEGIN_MARK, END_MARK, fixtures_markdown),
2853	            ("COOKING-TIMERS-AND-ALARMS.md", ALARM_BEGIN, ALARM_END, alarm_cases_markdown),
2854	            ("FIELD-QUALITY.md", REPORT_BEGIN, REPORT_END, report_example_markdown),
2855	            ("FIELD-QUALITY.md", CODES_BEGIN, CODES_END, event_codes_markdown)]
2856	    changed = 0
2857	    for name, begin, end, make in jobs:
2858	        path = os.path.join(eng, name)
2859	        if not os.path.exists(path):
2860	            print("skip (no file):", path)
2861	            continue
2862	        doc = open(path, encoding="utf-8").read()
2863	        if doc.count(begin) != 1 or doc.count(end) != 1 or doc.index(begin) > doc.index(end):
2864	            print("ERROR: %s must hold exactly one block from %r to %r" % (path, begin[:40], end[:40]))
2865	            return -1
2866	        i, j = doc.index(begin), doc.index(end) + len(end)
2867	        fresh = make().strip()
2868	        if doc[i:j].strip() != fresh:
2869	            with open(path, "w", encoding="utf-8") as fh:
2870	                fh.write(doc[:i] + fresh + doc[j:])
2871	            print("updated a generated block in", path)
2872	            changed += 1
2873	    return changed
